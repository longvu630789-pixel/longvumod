-- LOGIC SERVER HOP
-- =======================================================
local isBusy = false

TeleportService.TeleportInitFailed:Connect(function(player, teleportResult, errorMessage)
    if player == LocalPlayer then
        ActionBtn.Text = "Lỗi: " .. errorMessage
        ActionBtn.TextColor3 = Color3.fromRGB(255, 50, 50)
        task.wait(3)
        ActionBtn.Text = "HOP SERVER ÍT NGƯỜI"
        ActionBtn.TextColor3 = CONFIG.AccentColor
        isBusy = false
    end
end)

ActionBtn.MouseButton1Click:Connect(function()
    if isBusy then return end
    isBusy = true

    ActionBtn.Text = "Đang quét server..."
    ActionBtn.TextColor3 = Color3.fromRGB(255, 200, 0)

    local apiUrl = "https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
    local success, result = pcall(function() return game:HttpGet(apiUrl) end)

    if not success then
        ActionBtn.Text = "Lỗi kết nối API!"
        ActionBtn.TextColor3 = Color3.fromRGB(255, 50, 50)
        task.wait(2)
        ActionBtn.Text = "HOP SERVER ÍT NGƯỜI"
        ActionBtn.TextColor3 = CONFIG.AccentColor
        isBusy = false
        return
    end

    local data = HttpService:JSONDecode(result)
    local targetServer = nil
    local bestPlayerCount = 999

    if data and data.data then
        for _, server in ipairs(data.data) do
            if server.id ~= JobId then
                if server.playing <= 3 and server.playing < bestPlayerCount then
                    bestPlayerCount = server.playing
                    targetServer = server
                end
            end
        end
    end

    if targetServer then
        ActionBtn.Text = "Đang vào (" .. targetServer.playing .. " người)..."
        if targetServer.playing == 0 then
            ActionBtn.TextColor3 = Color3.fromRGB(100, 255, 100)
        else
            ActionBtn.TextColor3 = Color3.fromRGB(0, 200, 255)
        end

        local tpSuccess = pcall(function()
            TeleportService:TeleportToPlaceInstance(PlaceId, targetServer.id, LocalPlayer)
        end)

        if not tpSuccess then
            ActionBtn.Text = "Teleport bị chặn!"
            ActionBtn.TextColor3 = Color3.fromRGB(255, 50, 50)
            task.wait(2)
            ActionBtn.Text = "HOP SERVER ÍT NGƯỜI"
            ActionBtn.TextColor3 = CONFIG.AccentColor
            isBusy = false
        end
    else
        ActionBtn.Text = "Không tìm thấy server trống!"
        ActionBtn.TextColor3 = Color3.fromRGB(255, 50, 50)
        task.wait(2)
        ActionBtn.Text = "HOP SERVER ÍT NGƯỜI"
        ActionBtn.TextColor3 = CONFIG.AccentColor
        isBusy = false
    end
end)

-- TOGGLE & DRAG (NÚT TRÒN ẨN HIỆN)
-- =======================================================
local isOpen = false
local function ToggleMenu()
    isOpen = not isOpen
    MainFrame.Visible = true
    local targetSize = isOpen and CONFIG.MenuSize or UDim2.new(0, 0, 0, 0)
    TweenService:Create(MainFrame, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Size = targetSize}):Play()
    if not isOpen then
        task.delay(0.4, function()
            if not isOpen then MainFrame.Visible = false end
        end)
    end
end

CloseBtn.MouseButton1Click:Connect(function()
    if isOpen then ToggleMenu() end
end)

-- Kéo thả
local draggingToggle, dragInputToggle, dragStartToggle, startPosToggle
ToggleBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        draggingToggle = true
        dragStartToggle = input.Position
        startPosToggle = ToggleBtn.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then draggingToggle = false end
        end)
    end
end)
ToggleBtn.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInputToggle = input
    end
end)

local draggingMenu, dragInputMenu, dragStartMenu, startPosMenu
Title.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        draggingMenu = true
        dragStartMenu = input.Position
        startPosMenu = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then draggingMenu = false end
        end)
    end
end)
Title.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInputMenu = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInputToggle and draggingToggle then
        local delta = input.Position - dragStartToggle
        ToggleBtn.Position = UDim2.new(startPosToggle.X.Scale, startPosToggle.X.Offset + delta.X, startPosToggle.Y.Scale, startPosToggle.Y.Offset + delta.Y)
    elseif input == dragInputMenu and draggingMenu then
        local delta = input.Position - dragStartMenu
        MainFrame.Position = UDim2.new(startPosMenu.X.Scale, startPosMenu.X.Offset + delta.X, startPosMenu.Y.Scale, startPosMenu.Y.Offset + delta.Y)
    end
end)
